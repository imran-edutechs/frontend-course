Callback function
- Class function is function that can to anther function as a arguments

Examples:
function greet(name, callback) {
      console.log(`Hello, ${name}`);
      callback();
}

function afterGreeting() {
      console.log("Greeting finished!");
}

greet("Sakib", afterGreeting);


setTimeout(() => {
      console.log("2 seconds later");
}, 2000);


function fetchUser(callback) {
      console.log("Loading user...");

      setTimeout(() => {
            const user = {
                  name: "Sakib",
                  email: "sakib@example.com",
            };

            callback(user);
      }, 2000);
}

function showUserProfile(user) {
      console.log(`Name: ${user.name}`);
      console.log(`Email: ${user.email}`);
}

fetchUser(showUserProfile);




Scope in Javascript
- There are three type of scope in Javascript
1, Global
2, function
3, Block Scope


const appName = "Hazaro"; // Global scope

function showGame() {
      const gameName = "Friday Night Hazaro"; // Function scope

      if (true) {
            let playerName = "Sakib"; // Block scope

            console.log(appName);     // কাজ করবে
            console.log(gameName);    // কাজ করবে
            console.log(playerName);  // কাজ করবে
      }

      console.log(appName);     // কাজ করবে
      console.log(gameName);    // কাজ করবে
      // console.log(playerName); // Error
}

showGame();

console.log(appName); // কাজ করবে
// console.log(gameName); // Error

class User {
      constructor(name, email) {
            this.name = name;
            this.email = email;
      }

      login() {
            console.log(`${this.name} logged in`);
      }
}

const user1 = new User("Sakib", "sakib@example.com");
const user2 = new User("Rahim", "rahim@example.com");

user1.login(); // Sakib logged in
user2.login(); // Rahim logged in
